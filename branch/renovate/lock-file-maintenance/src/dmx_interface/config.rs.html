1//! Configuration data model for the DMX interface.
2//!
3//! Uses `heapless` fixed-size types for `no_std` compatibility. The JSON
4//! shape (field names, numeric enums, string-keyed index maps) mirrors
5//! `web/src/types/config.d.ts` — the WebSocket API is the contract between
6//! firmware and web UI; NVS keeps storing the same model via `postcard`.
7

11use serde::{Deserialize, Serialize};

13/// Maximum length for string configuration fields.
14const MAX_STR_LEN: usize = 32;

16/// Maximum number of DMX ports addressed by the config maps.
17const MAX_DMX_PORTS: usize = 4;

19/// Maximum number of button actions addressed by the config maps.
20const MAX_BUTTON_ACTIONS: usize = 4;

22/// Maximum size of the postcard-encoded config blob in NVS.
23const MAX_CONFIG_BYTES: usize = 512;

25/// Error for invalid enum discriminants arriving over the wire.
26#[derive(Debug)]
27pub struct InvalidValue(pub u8);
28
29impl fmt::Display for InvalidValue {
30    /// Render as `invalid enum value N`.
31    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
32        write!(f, "invalid enum value {}", self.0)
33    }
34}

36/// Network connection type (web: `ConnectionType`).
37#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
38#[serde(try_from = "u8", into = "u8")]
39pub enum ConnectionType {
40    /// Access Point mode (creates its own network).
41    WifiAp = 0,
42    /// Station mode (connects to an existing network).
43    WifiSta = 1,
44    /// Wired ethernet (not implemented yet).
45    Ethernet = 2,
46}
47
48impl Default for ConnectionType {
49    /// Devices boot as access point until configured otherwise.
50    fn default() -> Self {
51        Self::WifiAp
52    }
53}
54
55impl From<ConnectionType> for u8 {
56    /// `WifiAp` → 0, `WifiSta` → 1, `Ethernet` → 2.
57    fn from(value: ConnectionType) -> Self {
58        match value {
59            ConnectionType::WifiAp => 0,
60            ConnectionType::WifiSta => 1,
61            ConnectionType::Ethernet => 2,
62        }
63    }
64}
65
66impl TryFrom<u8> for ConnectionType {
67    /// Discriminant conversion failed.
68    type Error = InvalidValue;
69
70    /// Map a raw discriminant back to the enum.
71    fn try_from(value: u8) -> Result<Self, Self::Error> {
72        match value {
73            0 => Ok(Self::WifiAp),
74            1 => Ok(Self::WifiSta),
75            2 => Ok(Self::Ethernet),
76            other => Err(InvalidValue(other)),
77        }
78    }
79}

234/// `serde` adapter: index maps as JSON objects with decimal string keys.
235///
236/// The web UI addresses ports by object key (`dmx_ports["0"]`), while
237/// `serde-json-core` would emit integer keys as bare digits — invalid JSON
238/// that browsers reject. Serializing through `String<N>` and parsing the
239/// keys back keeps both directions valid.
240mod index_map {
241    use core::fmt::{self, Write as _};
242    use core::marker::PhantomData;
243
244    use heapless::{FnvIndexMap, String};
245    use serde::de::{self, MapAccess, Visitor};
246    use serde::ser::{SerializeMap, Serializer};
247    use serde::{Deserialize, Deserializer, Serialize};
248
249    /// Serialize `FnvIndexMap<u8, T, N>` as `{"0": …, "1": …}`.
250    pub fn serialize<S: Serializer, T: Serialize, const N: usize>(
251        map: &FnvIndexMap<u8, T, N>,
252        serializer: S,
253    ) -> Result<S::Ok, S::Error> {
254        /// Wrapper struct so `serde` serializes the map key as a string.
255        struct Key(u8);
256
257        impl Serialize for Key {
258            /// Write the decimal digits as a JSON string.
259            fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
260                let mut text = String::<4>::new();
261                write!(&mut text, "{}", self.0).map_err(serde::ser::Error::custom)?;
262                serializer.serialize_str(text.as_str())
263            }
264        }
265
266        let mut out = serializer.serialize_map(Some(map.len()))?;
267        for (key, value) in map {
268            out.serialize_entry(&Key(*key), value)?;
269        }
270        out.end()
271    }
272
273    /// Deserialize `{"0": …, "1": …}` back into `FnvIndexMap<u8, T, N>`.
274    pub fn deserialize<'de, D: Deserializer<'de>, T: Deserialize<'de>, const N: usize>(
275        deserializer: D,
276    ) -> Result<FnvIndexMap<u8, T, N>, D::Error> {
277        /// Visitor collecting string-keyed entries into the map.
278        struct MapVisitor<T, const N: usize>(PhantomData<T>);
279
280        impl<'de, T: Deserialize<'de>, const N: usize> Visitor<'de> for MapVisitor<T, N> {
281            /// The map produced by this visitor.
282            type Value = FnvIndexMap<u8, T, N>;
283
284            /// Human-readable description for parse errors.
285            fn expecting(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
286                f.write_str("a map with decimal string keys")
287            }
288
289            /// Parse each `"key": value` entry into the map.
290            fn visit_map<A: MapAccess<'de>>(self, mut access: A) -> Result<Self::Value, A::Error> {
291                let mut map = FnvIndexMap::new();
292                while let Some((key, value)) = access.next_entry::<String<8>, T>()? {
293                    let index: u8 = key.parse().map_err(de::Error::custom)?;
294                    map.insert(index, value)
295                        .map_err(|_| de::Error::custom("config map full"))?;
296                }
297                Ok(map)
298            }
299        }
300
301        deserializer.deserialize_map(MapVisitor(PhantomData))
302    }
303}

305/// Complete device configuration (web: `Config`).
306///
307/// Field suffixes mirror the web JSON contract — renaming for the lint
308/// would break the API.
309#[allow(clippy::struct_field_names)]
310#[derive(Debug, Clone, Serialize, Deserialize)]
311pub struct Config {
312    /// Configuration schema version (for future migration).
313    pub version: u8,
314    /// Network connection type.
315    pub connection: ConnectionType,
316    /// IP assignment method for station mode.
317    pub ip_method: IpMethod,
318    /// LED brightness (0–255).
319    pub led_brightness: u8,
320    /// `WiFi` station credentials.
321    pub station_config: WifiConfig,
322    /// `WiFi` access-point credentials (empty SSID derives from the MAC).
323    pub ap_config: WifiConfig,
324    /// DMX ports by port index.
325    #[serde(with = "index_map")]
326    pub dmx_ports: FnvIndexMap<u8, DmxPortConfig, MAX_DMX_PORTS>,
327    /// Button actions by button index.
328    #[serde(with = "index_map")]
329    pub button_actions: FnvIndexMap<u8, ButtonAction, MAX_BUTTON_ACTIONS>,
330}
331
332impl Default for Config {
333    /// Factory defaults: AP mode, DMX0 output on universe 1, DMX1 input on
334    /// universe 2, dim LED.
335    fn default() -> Self {
336        let ap_config = WifiConfig {
337            password: String::from("ChaosDMX"),
338            ..WifiConfig::default()
339        };
340        let mut dmx_ports = FnvIndexMap::new();
341        let _ = dmx_ports.insert(0, DmxPortConfig::default());
342        let _ = dmx_ports.insert(
343            1,
344            DmxPortConfig {
345                direction: DmxDirection::Input,
346                universe: 2,
347            },
348        );
349        Self {
350            version: 1,
351            connection: ConnectionType::default(),
352            ip_method: IpMethod::default(),
353            led_brightness: 25,
354            station_config: WifiConfig::default(),
355            ap_config,
356            dmx_ports,
357            button_actions: FnvIndexMap::new(),
358        }
359    }
360}

362impl Config {
363    /// Serialize config to bytes using postcard (binary format).
364    ///
365    /// Returns a `Vec` with max [`MAX_CONFIG_BYTES`] capacity.
366    pub fn to_bytes(&self) -> Result<Vec<u8, MAX_CONFIG_BYTES>, postcard::Error> {
367        postcard::to_vec(self)
368    }
369
370    /// Deserialize config from bytes.
371    pub fn from_bytes(data: &[u8]) -> Result<Self, postcard::Error> {
372        postcard::from_bytes(data)
373    }
374
375    /// Pushes this configuration into every driver that consumes it.
376    ///
377    /// Called once at boot after loading and again whenever the config
378    /// changes (WebSocket `config.set`, factory reset) — the single place
379    /// mapping config fields onto driver state. Network changes apply on
380    /// the next reboot.
381    pub fn apply(&self) {
382        crate::hardware::led::set_brightness(self.led_brightness);
383    }
384
385    /// Merge a partial patch onto this config (present fields win).
386    ///
387    /// This is the server-side counterpart of the web UI's
388    /// `DeepPartial<Config>` PATCH semantics: nested objects
389    /// (`ap_config`, `station_config`, `dmx_ports[*]`) merge field by
390    /// field, map entries merge per key, scalar fields replace when
391    /// present.
392    ///
393    /// Returns `false` when the patch is rejected — a map key outside
394    /// `0..MAX_*` or a map that is already full. Keys are validated before
395    /// anything is merged, so a rejected patch leaves the config untouched
396    /// (HTTP/WS answer 400 and must not persist).
397    pub fn merge_patch(&mut self, patch: ConfigPatch) -> bool {
398        if patch
399            .dmx_ports
400            .keys()
401            .any(|&key| usize::from(key) >= MAX_DMX_PORTS)
402            || patch
403                .button_actions
404                .keys()
405                .any(|&key| usize::from(key) >= MAX_BUTTON_ACTIONS)
406        {
407            return false;
408        }
409        if let Some(version) = patch.version {
410            self.version = version;
411        }
412        if let Some(connection) = patch.connection {
413            self.connection = connection;
414        }
415        if let Some(ip_method) = patch.ip_method {
416            self.ip_method = ip_method;
417        }
418        if let Some(led_brightness) = patch.led_brightness {
419            self.led_brightness = led_brightness;
420        }
421        if let Some(station_config) = patch.station_config {
422            station_config.apply(&mut self.station_config);
423        }
424        if let Some(ap_config) = patch.ap_config {
425            ap_config.apply(&mut self.ap_config);
426        }
427        for (key, port_patch) in patch.dmx_ports {
428            let mut port = self.dmx_ports.get(&key).copied().unwrap_or_default();
429            port_patch.apply(&mut port);
430            if self.dmx_ports.insert(key, port).is_err() {
431                return false;
432            }
433        }
434        for (key, action) in patch.button_actions {
435            if self.button_actions.insert(key, action).is_err() {
436                return false;
437            }
438        }
439        true
440    }
441}

443/// One [`WifiConfig`] entry as it arrives in a patch (web: `DeepPartial`).
444///
445/// The UI edits credentials leaf by leaf, so both fields are optional
446/// and absent fields keep their current value.
447#[derive(Debug, Default, Deserialize)]
448pub struct WifiConfigPatch {
449    /// Optional network name.
450    pub ssid: Option<String<MAX_STR_LEN>>,
451    /// Optional password.
452    pub password: Option<String<MAX_STR_LEN>>,
453}
454
455impl WifiConfigPatch {
456    /// Move the present fields onto `target`; absent fields stay untouched.
457    fn apply(self, target: &mut WifiConfig) {
458        if let Some(ssid) = self.ssid {
459            target.ssid = ssid;
460        }
461        if let Some(password) = self.password {
462            target.password = password;
463        }
464    }
465}

488/// Partial configuration (web: `DeepPartial<Config>`).
489///
490/// Every field is optional — including nested objects: credentials and
491/// DMX-port entries arrive leaf by leaf (the UI only sends changed
492/// leaves) and merge field-wise onto the stored values.
493#[derive(Debug, Default, Deserialize)]
494pub struct ConfigPatch {
495    /// Optional schema version.
496    pub version: Option<u8>,
497    /// Optional connection type.
498    pub connection: Option<ConnectionType>,
499    /// Optional IP method.
500    pub ip_method: Option<IpMethod>,
501    /// Optional LED brightness.
502    pub led_brightness: Option<u8>,
503    /// Optional station credentials (fields merge individually).
504    pub station_config: Option<WifiConfigPatch>,
505    /// Optional access-point credentials (fields merge individually).
506    pub ap_config: Option<WifiConfigPatch>,
507    /// Optional per-port overrides (merged per key and field).
508    #[serde(default, with = "index_map")]
509    pub dmx_ports: FnvIndexMap<u8, DmxPortPatch, MAX_DMX_PORTS>,
510    /// Optional per-button overrides (merged per key).
511    #[serde(default, with = "index_map")]
512    pub button_actions: FnvIndexMap<u8, ButtonAction, MAX_BUTTON_ACTIONS>,
513}

515/// Serialization round-trip tests.
516#[cfg(test)]
517mod tests {
518    use super::*;
519
520    /// Encode → decode must preserve the important fields.
521    #[test]
522    fn config_roundtrip() {
523        let config = Config::default();
524        let bytes = config.to_bytes().unwrap();
525        let restored = Config::from_bytes(&bytes).unwrap();
526        assert_eq!(config.connection, restored.connection);
527        assert_eq!(config.led_brightness, restored.led_brightness);
528        assert_eq!(
529            config.dmx_ports.get(&0).map(|port| port.universe),
530            restored.dmx_ports.get(&0).map(|port| port.universe)
531        );
532    }
