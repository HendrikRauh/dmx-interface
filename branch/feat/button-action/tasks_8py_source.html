11def _find_esp_port():
12 """Finds the first matching USB port based on known ESP VIDs/PIDs."""
13 # Known IDs for ESP boards and common USB-UART bridges
14 ESP_IDENTIFIERS = [
15 (0x303A, None), # Espressif (e.g., ESP32-S3/C3 native USB-JTAG-Serial)
16 (0x10C4, 0xEA60), # Silicon Labs CP210x (very common on ESP32)
17 (0x1A86, 0x7523), # WCH CH340 (common on budget ESP boards)
18 (0x0403, 0x6001), # FTDI FT232R
19 ]
20
21 ports = serial.tools.list_ports.comports()
22
23 for port in ports:
24 # Check if VID/PID match
25 for vid, pid in ESP_IDENTIFIERS:
26 if port.vid == vid and (pid is None or port.pid == pid):
27 return port.device
28
29 # Fallback to default if no matching device is found
30 return "/dev/ttyUSB0"
