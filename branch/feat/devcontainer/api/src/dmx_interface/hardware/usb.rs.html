1//! USB-CDC-ACM device with esptool auto-reset support.
2//!
3//! The Lolin S2 Mini has no USB-UART bridge, so esptool cannot toggle
4//! GPIO0/EN via physical DTR/RTS pins. Instead, the firmware exposes the
5//! native USB-OTG port as a CDC-ACM device and interprets the CDC
6//! control-line state the way the classic devkit reset sequence does:
7//!
8//! | DTR | RTS | Meaning            | Action                          |
9//! |-----|-----|--------------------|---------------------------------|
10//! | 1   | 0   | IO0 low            | set `FORCE_DOWNLOAD_BOOT`       |
11//! | 0   | 1   | EN low (reset)     | software reset                  |
12//! | 0   | 0   | idle               | clear `FORCE_DOWNLOAD_BOOT`     |
13//! | 1   | 1   | debounced clear    | clear `FORCE_DOWNLOAD_BOOT`     |
14//!
15//! esptool's `--before usb-reset` walk is `(1,0)` (arm) -> a ~1-3 ms
16//! `(1,1)` blip -> `(0,1)` (reset): the `(0,1)` step boots whichever
17//! target `FORCE_DOWNLOAD_BOOT` selects. The `(1,1)` step is the hard
18//! case for this state machine, because a terminal program also ends up
19//! in `(1,1)`: pyserial asserts DTR and RTS with **separate** ioctls on
20//! open, so the firmware sees a `(1,0)` pulse before `(1,1)` — which
21//! would leave the flag armed for the lifetime of the connection, and
22//! any later reset (panic, `inv monitor`'s own open dance after a device
23//! loss) would land in the bootloader instead of the app. So `(1,1)`
24//! clears the flag, but only after it persists for [`OPEN_DEBOUNCE`]
25//! (20 ms): long enough to outlive a terminal's session-long hold, far
26//! longer than the walk's blip — which therefore never disarms the flag
27//! mid-walk.
28//!
29//! The flag lives in the RTC domain and survives software resets. The ROM
30//! bootloader does **not** clear it on entry — after a flash the chip would
31//! stay stuck in the bootloader forever (until power-on). `inv flash`
32//! therefore clears the flag itself after flashing (esptool `write-reg
33//! 0x3F408128 0 0x1` through the stub) and then resets via the RTC watchdog,
34//! which boots the application.
35//!
36//! The same task also drains the log ring from [`crate::logging`] over the
37//! CDC data endpoint, so `inv monitor` shows the firmware log output.
38

59/// Espressif USB vendor ID (same as the ROM bootloader).
60const USB_VID: u16 = 0x303A;
61
62/// Espressif generic CDC-ACM product ID.
63const USB_PID: u16 = 0x3001;
64
65/// Line-poll interval of the DTR/RTS state machine.
66const POLL_INTERVAL: Duration = Duration::from_millis(10);
67
68/// How long `(1,1)` must persist before it disarms `FORCE_DOWNLOAD_BOOT`.
69///
70/// pyserial holds `(1,1)` for the whole session (clearing takes effect
71/// after this window), while esptool's walk only blips it for ~1-3 ms —
72/// shorter than the window, so a flash is never disarmed mid-walk.
73const OPEN_DEBOUNCE: Duration = Duration::from_millis(20);
74
75/// Log chunk size handed to the CDC endpoint (matches `max_packet_size`).
76const LOG_CHUNK: usize = 64;
77
78/// The S2 peripheral types are invariant in their lifetime, which pins the
79/// USB driver to `'static`; the embassy-usb buffers therefore must be
80/// `'static` too. They are initialized exactly once at task start.
81static EP_OUT_BUFFER: StaticCell<[u8; 1024]> = StaticCell::new();
82/// Bulk OUT endpoint buffer for the CDC data endpoint.
83static CONFIG_DESCRIPTOR: StaticCell<[u8; 256]> = StaticCell::new();
84/// Buffer for the generated USB configuration descriptor.
85static BOS_DESCRIPTOR: StaticCell<[u8; 256]> = StaticCell::new();
86/// Buffer for the generated USB BOS descriptor.
87static CONTROL_BUF: StaticCell<[u8; 64]> = StaticCell::new();
88/// Scratch buffer for USB control transfers.
89static CDC_STATE: StaticCell<State<'static>> = StaticCell::new();

91/// Runs the USB-CDC-ACM device and the esptool reset protocol.
92///
93/// The DP/DM pins are fixed by the S2 hardware (GPIO20/GPIO19).
94#[embassy_executor::task]
95pub async fn task(usb0: USB0<'static>, dp: GPIO20<'static>, dm: GPIO19<'static>) -> ! {
96    let usb = Usb::new(usb0, dp, dm);
97
98    let driver = Driver::new(usb, EP_OUT_BUFFER.init([0u8; 1024]), Config::default());
99
100    let mut config = embassy_usb::Config::new(USB_VID, USB_PID);
101    config.manufacturer = Some("dmx-interface");
102    config.product = Some("DMX-Interface USB-CDC");
103    config.device_class = 0xEF;
104    config.device_sub_class = 0x02;
105    config.device_protocol = 0x01;
106    config.composite_with_iads = true;
107
108    let mut builder = Builder::new(
109        driver,
110        config,
111        CONFIG_DESCRIPTOR.init([0u8; 256]),
112        BOS_DESCRIPTOR.init([0u8; 256]),
113        &mut [],
114        CONTROL_BUF.init([0u8; 64]),
115    );
116    let mut class = CdcAcmClass::new(&mut builder, CDC_STATE.init(State::new()), 64);
117    let mut device = builder.build();
118
119    log::info!("USB-CDC ready");
120    join(device.run(), control_and_log(&mut class)).await;
121    unreachable!()
122}

124/// Runs the DTR/RTS state machine and drains the log ring into the host.
125///
126/// Both roles live in one task because the line poll needs `&class` while
127/// the log flush needs `&mut class` — a single `&mut` serves both (`dtr()`
128/// and `rts()` take `&self`).
129///
130/// The future is a `select` of two branches:
131///
132/// * a deadline-based 10 ms timer that re-checks the control lines (the
133///   deadline is only re-armed when the timer actually fired, so log traffic
134///   can never starve the esptool reset walk), and
135/// * a log flush that copies one chunk out of the ring and awaits
136///   `write_packet`. Bytes are only committed after the driver accepted the
137///   packet, so a cancelled flush (timer branch won) loses nothing and
138///   duplicates nothing.
139///
140/// The flush is gated on DTR: the host's CDC driver (Linux cdc-acm, Windows
141/// usbser) binds and submits receive URBs as soon as the device is
142/// configured — long before anyone opens the tty. Draining into that sink
143/// would swallow boot logs and the panic replay before `inv monitor` gets a
144/// chance to read them (the tty layer discards buffered input on open). With
145/// the gate, bytes stay in the ring until a host asserts DTR, which every
146/// terminal program does on open.
147async fn control_and_log(class: &mut CdcAcmClass<'static, Driver<'static>>) -> ! {
148    let mut prev = (class.dtr(), class.rts());
149    let mut deadline = Instant::now().saturating_add(POLL_INTERVAL);
150    let mut open_since: Option<Instant> = None;
151    let mut open_cleared = false;
152    loop {
153        let outcome = select(Timer::at(deadline), flush_logs(class)).await;
154        match outcome {
155            Either::First(()) => {
156                deadline = Instant::now().saturating_add(POLL_INTERVAL);
157                let now = (class.dtr(), class.rts());
158                if now != prev {
159                    prev = now;
160                    open_since = None;
161                    open_cleared = false;
162                    match now {
163                        // `(false,false)` idle: disarm immediately. `(true,true)`
164                        // starts the debounce instead of clearing (see
165                        // [`OPEN_DEBOUNCE`]) — a terminal holds it for the whole
166                        // session, esptool's walk only blips it.
167                        (false, false) => set_force_download(false),
168                        (true, true) => open_since = Some(Instant::now()),
169                        // pyserial asserts DTR before RTS: arm on the `(1,0)`
170                        // transient, then either the debounce (terminal) or the
171                        // `(0,1)` reset (esptool) follows.
172                        (true, false) => set_force_download(true),
173                        (false, true) => enter_bootloader(),
174                    }
175                }
176                if prev == (true, true)
177                    && !open_cleared
178                    && open_since.is_some_and(|since| since.elapsed() >= OPEN_DEBOUNCE)
179                {
180                    set_force_download(false);
181                    open_cleared = true;
182                }
183            }
184            Either::Second(()) => {}
185        }
186    }
187}

189/// Copies pending log lines from the ring into the CDC endpoint until the
190/// ring is empty or the endpoint errors out. See [`control_and_log`] for the
191/// cancellation-safety argument and the DTR gate rationale.
192async fn flush_logs(class: &mut CdcAcmClass<'static, Driver<'static>>) {
193    if !class.dtr() {
194        // No terminal attached: keep the ring intact (drop-oldest still
195        // bounds it) and idle for one poll interval.
196        Timer::after(POLL_INTERVAL).await;
197        return;
198    }
199    loop {
200        let mut chunk = [0u8; LOG_CHUNK];
201        let n = logging::cdc_peek(&mut chunk);
202        if n == 0 {
203            Timer::after(POLL_INTERVAL).await;
204            return;
205        }
206        let (data, _) = chunk.split_at(n);
207        if class.write_packet(data).await.is_err() {
208            // The endpoint is disabled (host has not configured us yet) —
209            // returning immediately would let `select` complete over and
210            // over without ever yielding, starving `device.run()` and
211            // stalling enumeration. Wait instead of spinning.
212            Timer::after(POLL_INTERVAL).await;
213            return;
214        }
215        logging::cdc_commit(n);
216    }
217}

219/// Set or clear the ROM `FORCE_DOWNLOAD_BOOT` flag in `RTC_CNTL_OPTION1`.
220///
221/// The register lives in the RTC domain and survives a software reset, so
222/// the ROM bootloader sees it on the next boot.
223fn set_force_download(armed: bool) {
224    // SAFETY: RTC registers are always memory-mapped; `steal()` only bypasses
225    // singleton ownership of the peripheral and `modify` leaves every other
226    // bit of `RTC_CNTL_OPTION1` untouched.
227    let rtc_cntl = unsafe { esp32s2::RTC_CNTL::steal() };
228    rtc_cntl
229        .options1()
230        .modify(|_, w| w.force_download_boot().bit(armed));
231}

233/// Resets into the ROM USB-CDC download mode (bootloader).
234///
235/// `(0,1)` is a plain reset: the boot target is decided by the
236/// `FORCE_DOWNLOAD_BOOT` flag, which the `(1,0)` state arms. esptool's
237/// `--before usb-reset` walk therefore lands in the bootloader. The ROM
238/// bootloader does not clear the flag itself — it is cleared by
239/// `inv flash` after flashing (esptool `write-reg`), by `(0,0)`/`(1,1)`
240/// line states, or by the terminal-open disarming above.
241fn enter_bootloader() -> ! {
242    esp_hal::system::software_reset()
243}
