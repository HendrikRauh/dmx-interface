129static esp_err_t get_config_handler(httpd_req_t *req) {
130 LOGI("Received GET request for /api/config");
131
132 cJSON *root = cJSON_CreateObject();
133 if (!root) {
134 goto fail;
135 }
136
137 // --- Root config fields ---
138 cJSON_AddNumberToObject(root, KEY_CONNECTION, config_get_connection());
139 cJSON_AddNumberToObject(root, KEY_IP_METHOD, config_get_ip_method());
140 cJSON_AddNumberToObject(root, KEY_LED_BRIGHTNESS,
141 config_get_led_brightness());
142
143 wifi_config_t wifi_config;
144 char buf_ssid[33] = {0};
145 char buf_pass[65] = {0};
146
147 // --- WiFi Station Config ---
148 config_get_wifi_sta_config(&wifi_config);
149 cJSON *json_wifi_sta = cJSON_CreateObject();
150 if (!json_wifi_sta) {
151 goto fail;
152 }
153 cJSON_AddItemToObject(root, KEY_WIFI_STATION_CONFIG, json_wifi_sta);
154 memcpy(buf_ssid, wifi_config.sta.ssid, sizeof(wifi_config.sta.ssid));
155 memcpy(buf_pass, wifi_config.sta.password, sizeof(wifi_config.sta.password));
156 cJSON_AddStringToObject(json_wifi_sta, KEY_SSID, buf_ssid);
157 cJSON_AddStringToObject(json_wifi_sta, KEY_PASSWORD, buf_pass);
158
159 // --- WiFi AP Config ---
160 config_get_wifi_ap_config(&wifi_config);
161 cJSON *json_wifi_ap = cJSON_CreateObject();
162 if (!json_wifi_ap) {
163 goto fail;
164 }
165 cJSON_AddItemToObject(root, KEY_WIFI_AP_CONFIG, json_wifi_ap);
166 memset(buf_ssid, 0, sizeof(buf_ssid));
167 memset(buf_pass, 0, sizeof(buf_pass));
168 memcpy(buf_ssid, wifi_config.ap.ssid, sizeof(wifi_config.ap.ssid));
169 memcpy(buf_pass, wifi_config.ap.password, sizeof(wifi_config.ap.password));
170 cJSON_AddStringToObject(json_wifi_ap, KEY_SSID, buf_ssid);
171 cJSON_AddStringToObject(json_wifi_ap, KEY_PASSWORD, buf_pass);
172
173 // --- DMX Ports Array ---
174 cJSON *dmx_array = cJSON_AddArrayToObject(root, KEY_DMX_PORTS);
175 if (!dmx_array) {
176 goto fail;
177 }
178 for (size_t i = 0; i < APP_CONFIG_DMX_PORT_COUNT; i++) {
179 cJSON *port_config = cJSON_CreateObject();
180 if (port_config) {
181 cJSON_AddItemToArray(dmx_array, port_config);
182 cJSON_AddNumberToObject(port_config, KEY_DMX_PORT_INDEX, i);
183 cJSON_AddNumberToObject(port_config, KEY_UNIVERSE,
184 config_get_dmx_universe(i));
185 cJSON_AddNumberToObject(port_config, KEY_DMX_DIRECTION,
186 config_get_dmx_direction(i));
187 }
188 }
189
190 // Generate response string
191 char *json_str = cJSON_PrintUnformatted(root);
192 if (!json_str) {
193 goto fail;
194 }
195
196 cJSON_Delete(root);
197
198 httpd_resp_set_type(req, "application/json");
199 httpd_resp_sendstr(req, json_str);
200
201 cJSON_free(json_str);
202 return ESP_OK;
203
204fail:
205 cJSON_Delete(root);
206
207 LOGE("Failed to create JSON response due to memory constraints");
208 httpd_resp_set_status(req, "507 Insufficient Storage");
209 httpd_resp_send(req, NULL, 0);
210 return ESP_OK;
211}

248static void action_set_port_config(const cJSON *array) {
249 int len = cJSON_GetArraySize(array);
250 LOGI("Setting config for %d DMX ports", len);
251
252 for (int i = 0; i < len; i++) {
253 cJSON *item = cJSON_GetArrayItem(array, i);
254 const cJSON *port_index =
255 cJSON_GetObjectItemCaseSensitive(item, KEY_DMX_PORT_INDEX);
256 const cJSON *universe =
257 cJSON_GetObjectItemCaseSensitive(item, KEY_UNIVERSE);
258 const cJSON *direction =
259 cJSON_GetObjectItemCaseSensitive(item, KEY_DMX_DIRECTION);
260 LOGI("Port config - index: %d, universe: %d, direction: %d",
261 port_index->valueint, universe ? universe->valueint : -1,
262 direction ? direction->valueint : -1);
263
264 if (universe != NULL) {
265 config_set_dmx_universe(port_index->valueint, universe->valueint);
266 }
267
268 if (direction != NULL) {
269 config_set_dmx_direction(port_index->valueint, direction->valueint);
270 }
271 }
272}

315static esp_err_t post_config_handler(httpd_req_t *req) {
316 LOGI("Received POST request for /api/config");
317
318 cJSON *json;
319 if (get_json_body(req, &json) != ESP_OK) {
320 // Error response already sent by get_json_body
321 return ESP_OK;
322 }
323
324 char *error_msg = NULL;
325 esp_err_t validation_result =
326 json_processor_process(json, &processor, &error_msg);
327
328 cJSON_Delete(json);
329
330 if (validation_result != ESP_OK) {
331 if (validation_result == ESP_ERR_NO_MEM) {
332 LOGE("Memory allocation failed while processing JSON");
333 httpd_resp_set_status(req, "507 Insufficient Storage");
334 httpd_resp_send(req, NULL, 0);
335 return ESP_OK;
336 }
337
338 if (validation_result == ESP_ERR_INVALID_ARG) {
339 LOGW("Received invalid JSON data in POST /api/config: %s", error_msg);
340 httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, error_msg);
341 free(error_msg);
342 return ESP_OK;
343 }
344
345 LOGE("Unexpected error while processing JSON: %s", error_msg);
346 httpd_resp_set_status(req, "500 Internal Server Error");
347 httpd_resp_send(req, NULL, 0);
348 free(error_msg);
349 return ESP_OK;
350 }
351
352 httpd_resp_send(req, NULL, 0);
353 return ESP_OK;
354}
