175static esp_err_t get_config_handler(httpd_req_t *req) {
176 LOGI("Received GET request for /api/config");
177
178 cJSON *root = cJSON_CreateObject();
179 if (!root) {
180 goto fail;
181 }
182
183 // --- Root config fields ---
184 cJSON_AddNumberToObject(root, KEY_CONNECTION, config_get_connection());
185 cJSON_AddNumberToObject(root, KEY_IP_METHOD, config_get_ip_method());
186 cJSON_AddNumberToObject(root, KEY_LED_BRIGHTNESS,
187 config_get_led_brightness());
188
189 wifi_config_t wifi_config;
190 char buf_ssid[33] = {0};
191 char buf_pass[65] = {0};
192
193 // --- WiFi Station Config ---
194 config_get_wifi_sta_config(&wifi_config);
195 cJSON *json_wifi_sta = cJSON_CreateObject();
196 if (!json_wifi_sta) {
197 goto fail;
198 }
199 cJSON_AddItemToObject(root, KEY_WIFI_STATION_CONFIG, json_wifi_sta);
200 memcpy(buf_ssid, wifi_config.sta.ssid, sizeof(wifi_config.sta.ssid));
201 memcpy(buf_pass, wifi_config.sta.password, sizeof(wifi_config.sta.password));
202 cJSON_AddStringToObject(json_wifi_sta, KEY_SSID, buf_ssid);
203 cJSON_AddStringToObject(json_wifi_sta, KEY_PASSWORD, buf_pass);
204
205 // --- WiFi AP Config ---
206 config_get_wifi_ap_config(&wifi_config);
207 cJSON *json_wifi_ap = cJSON_CreateObject();
208 if (!json_wifi_ap) {
209 goto fail;
210 }
211 cJSON_AddItemToObject(root, KEY_WIFI_AP_CONFIG, json_wifi_ap);
212 memset(buf_ssid, 0, sizeof(buf_ssid));
213 memset(buf_pass, 0, sizeof(buf_pass));
214 memcpy(buf_ssid, wifi_config.ap.ssid, sizeof(wifi_config.ap.ssid));
215 memcpy(buf_pass, wifi_config.ap.password, sizeof(wifi_config.ap.password));
216 cJSON_AddStringToObject(json_wifi_ap, KEY_SSID, buf_ssid);
217 cJSON_AddStringToObject(json_wifi_ap, KEY_PASSWORD, buf_pass);
218
219 // --- DMX Ports Array ---
220 cJSON *dmx_ports = cJSON_AddObjectToObject(root, KEY_DMX_PORTS);
221 if (!dmx_ports) {
222 goto fail;
223 }
224 for (size_t i = 0; i < APP_CONFIG_DMX_PORT_COUNT; i++) {
225 cJSON *port_config = cJSON_CreateObject();
226 if (port_config) {
227 char port_index_str[3];
228 sprintf(port_index_str, "%d", i);
229 cJSON_AddItemToObject(dmx_ports, port_index_str, port_config);
230 cJSON_AddNumberToObject(port_config, KEY_UNIVERSE,
231 config_get_dmx_universe(i));
232 cJSON_AddNumberToObject(port_config, KEY_DMX_DIRECTION,
233 config_get_dmx_direction(i));
234 }
235 }
236
237 // Generate response string
238 char *json_str = cJSON_PrintUnformatted(root);
239 if (!json_str) {
240 goto fail;
241 }
242
243 cJSON_Delete(root);
244
245 httpd_resp_set_type(req, "application/json");
246 httpd_resp_sendstr(req, json_str);
247
248 cJSON_free(json_str);
249 return ESP_OK;
250
251fail:
252 cJSON_Delete(root);
253
254 LOGE("Failed to create JSON response due to memory constraints");
255 httpd_resp_set_status(req, "507 Insufficient Storage");
256 httpd_resp_send(req, NULL, 0);
257 return ESP_OK;
258}

295static void action_set_port_config(const cJSON *ports) {
296 const cJSON *port = NULL;
297 cJSON_ArrayForEach(port, ports) {
298 const int port_index = atoi(port->string);
299 const cJSON *universe =
300 cJSON_GetObjectItemCaseSensitive(port, KEY_UNIVERSE);
301 const cJSON *direction =
302 cJSON_GetObjectItemCaseSensitive(port, KEY_DMX_DIRECTION);
303 LOGI("Port config - index: %d, universe: %d, direction: %d", port_index,
304 universe ? universe->valueint : -1,
305 direction ? direction->valueint : -1);
306
307 if (universe != NULL) {
308 config_set_dmx_universe(port_index, universe->valueint);
309 }
310
311 if (direction != NULL) {
312 config_set_dmx_direction(port_index, direction->valueint);
313 }
314 }
315}

358static esp_err_t post_config_handler(httpd_req_t *req) {
359 LOGI("Received POST request for /api/config");
360
361 cJSON *json;
362 if (get_json_body(req, &json) != ESP_OK) {
363 // Error response already sent by get_json_body
364 return ESP_OK;
365 }
366
367 char *error_msg = NULL;
368 esp_err_t validation_result =
369 json_processor_process(json, &processor, &error_msg);
370
371 cJSON_Delete(json);
372
373 if (validation_result != ESP_OK) {
374 if (validation_result == ESP_ERR_NO_MEM) {
375 LOGE("Memory allocation failed while processing JSON");
376 httpd_resp_set_status(req, "507 Insufficient Storage");
377 httpd_resp_send(req, NULL, 0);
378 return ESP_OK;
379 }
380
381 if (validation_result == ESP_ERR_INVALID_ARG) {
382 LOGW("Received invalid JSON data in POST /api/config: %s", error_msg);
383 httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, error_msg);
384 free(error_msg);
385 return ESP_OK;
386 }
387
388 LOGE("Unexpected error while processing JSON: %s", error_msg);
389 httpd_resp_set_status(req, "500 Internal Server Error");
390 httpd_resp_send(req, NULL, 0);
391 free(error_msg);
392 return ESP_OK;
393 }
394
395 httpd_resp_send(req, NULL, 0);
396 return ESP_OK;
397}
