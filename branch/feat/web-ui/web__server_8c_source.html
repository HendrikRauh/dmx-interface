56static const char *get_mime_type(const char *filename) {
57 const char *dot = strrchr(filename, '.');
58 if (!dot)
59 return "application/octet-stream";
60
61 if (strcmp(dot, ".html") == 0)
62 return "text/html";
63 if (strcmp(dot, ".css") == 0)
64 return "text/css";
65 if (strcmp(dot, ".js") == 0)
66 return "application/javascript";
67 if (strcmp(dot, ".json") == 0)
68 return "application/json";
69 if (strcmp(dot, ".png") == 0)
70 return "image/png";
71 if (strcmp(dot, ".jpg") == 0 || strcmp(dot, ".jpeg") == 0)
72 return "image/jpeg";
73 if (strcmp(dot, ".gif") == 0)
74 return "image/gif";
75 if (strcmp(dot, ".svg") == 0)
76 return "image/svg+xml";
77 if (strcmp(dot, ".ico") == 0)
78 return "image/x-icon";
79 if (strcmp(dot, ".txt") == 0)
80 return "text/plain";
81 if (strcmp(dot, ".xml") == 0)
82 return "application/xml";
83 if (strcmp(dot, ".wav") == 0)
84 return "audio/wav";
85 if (strcmp(dot, ".mp3") == 0)
86 return "audio/mpeg";
87
88 return "application/octet-stream";
89}

94static esp_err_t static_file_handler(httpd_req_t *req) {
95 // Build the file path
96 char filepath[1024];
97 snprintf(filepath, sizeof(filepath), "%s%s", storage_get_mount_point(),
98 req->uri);
99
100 // Handle root path
101 if (strcmp(req->uri, "/") == 0) {
102 snprintf(filepath, sizeof(filepath), "%s/index.html",
103 storage_get_mount_point());
104 }
105
106 FILE *f = fopen(filepath, "r");
107 if (!f) {
108 LOGW("File not found: %s", filepath);
109 httpd_resp_send_404(req);
110 return ESP_OK;
111 }
112
113 // Get MIME type and set content type
114 const char *mime_type = get_mime_type(filepath);
115 httpd_resp_set_type(req, mime_type);
116
117 // Send file in chunks
118 char buf[1024];
119 size_t read_len;
120 while ((read_len = fread(buf, 1, sizeof(buf), f)) > 0) {
121 if (httpd_resp_send_chunk(req, buf, read_len) != ESP_OK) {
122 LOGW("Failed to send data chunk for %s", filepath);
123 break;
124 }
125 }
126
127 fclose(f);
128 httpd_resp_send_chunk(req, NULL, 0); // Send end marker
129 return ESP_OK;
130}

158httpd_handle_t webserver_start(const webserver_config_t *config) {
159 if (s_server_handle != NULL) {
160 LOGW("Web server already running");
161 return s_server_handle;
162 }
163
164 // Initialize LittleFS
165 esp_err_t ret = storage_init();
166 if (ret != ESP_OK) {
167 LOGE("Failed to initialize storage");
168 return NULL;
169 }
170
171 // Use provided config or defaults
172 uint16_t port = WEBSERVER_DEFAULT_PORT;
173 size_t max_handlers = WEBSERVER_DEFAULT_MAX_HANDLERS;
174 size_t stack_size = WEBSERVER_DEFAULT_STACK_SIZE;
175 UBaseType_t task_priority = WEBSERVER_DEFAULT_TASK_PRIORITY;
176
177 if (config) {
178 port = config->port;
179 max_handlers = config->max_uri_handlers;
180 stack_size = config->stack_size;
181 task_priority = config->task_priority;
182 }
183
184 // Create HTTP server configuration
185 httpd_config_t http_config = HTTPD_DEFAULT_CONFIG();
186 http_config.server_port = port;
187 http_config.max_uri_handlers = max_handlers;
188 http_config.stack_size = stack_size;
189 http_config.uri_match_fn = httpd_uri_match_wildcard;
190
191 // Start HTTP server
192 ret = httpd_start(&s_server_handle, &http_config);
193 if (ret != ESP_OK) {
194 LOGE("Failed to start HTTP server: %s", esp_err_to_name(ret));
195 s_server_handle = NULL;
196 return NULL;
197 }
198
199 LOGI("HTTP server started on port %d", port);
200
201 const httpd_uri_t *routes = NULL;
202 size_t route_count = 0;
203
204 // Register config routes
205 route_count = get_config_routes(&routes);
206 webserver_register_handler_array(s_server_handle, routes, route_count);
207
208 // Wildcard handler for static files from LittleFS (must be last)
209 httpd_uri_t file_uri = {
210 .uri = "/*",
211 .method = HTTP_GET,
212 .handler = static_file_handler,
213 .user_ctx = NULL,
214 };
215 httpd_register_uri_handler(s_server_handle, &file_uri);
216
217 // Create FreeRTOS task for the server
218 // This allows other tasks to continue running and makes the server
219 // async-ready
220 BaseType_t task_ret = xTaskCreate(webserver_task, "webserver", stack_size,
221 (void *)s_server_handle, task_priority,
222 &s_server_task_handle);
223
224 if (task_ret != pdPASS) {
225 LOGE("Failed to create web server task");
226 httpd_stop(s_server_handle);
227 s_server_handle = NULL;
228 return NULL;
229 }
230
231 LOGI("Web server initialized successfully");
232 return s_server_handle;
233}
