112static esp_err_t static_file_handler(httpd_req_t *req) {
113 const static_file_t *file = get_file_data(req->uri);
114
115 if (file == NULL) {
116 LOGW("File not found: %s", req->uri);
117 httpd_resp_send_404(req);
118 return ESP_OK;
119 }
120
121 httpd_resp_set_type(req, file->content_type);
122 if (file->content_encoding) {
123 httpd_resp_set_hdr(req, "Content-Encoding", file->content_encoding);
124 }
125
126 const size_t file_len = file->end - file->start;
127 httpd_resp_send(req, (const char *)file->start, file_len);
128
129 return ESP_OK;
130}

132httpd_handle_t webserver_start(const webserver_config_t *config) {
133 if (s_server_handle != NULL) {
134 LOGW("Web server already running");
135 return s_server_handle;
136 }
137
138 // Use provided config or defaults
139 uint16_t port = WEBSERVER_DEFAULT_PORT;
140 size_t max_handlers = WEBSERVER_DEFAULT_MAX_HANDLERS;
141 size_t stack_size = WEBSERVER_DEFAULT_STACK_SIZE;
142 UBaseType_t task_priority = WEBSERVER_DEFAULT_TASK_PRIORITY;
143
144 if (config) {
145 port = config->port;
146 max_handlers = config->max_uri_handlers;
147 stack_size = config->stack_size;
148 task_priority = config->task_priority;
149 }
150
151 // Create HTTP server configuration
152 httpd_config_t http_config = HTTPD_DEFAULT_CONFIG();
153 http_config.server_port = port;
154 http_config.max_uri_handlers = max_handlers;
155 http_config.stack_size = stack_size;
156 http_config.uri_match_fn = httpd_uri_match_wildcard;
157 http_config.task_priority = task_priority;
158
159 // Start HTTP server
160 esp_err_t ret = httpd_start(&s_server_handle, &http_config);
161 if (ret != ESP_OK) {
162 LOGE("Failed to start HTTP server: %s", esp_err_to_name(ret));
163 s_server_handle = NULL;
164 return NULL;
165 }
166
167 LOGI("HTTP server started on port %d", port);
168
169 const httpd_uri_t *routes = NULL;
170 size_t route_count = 0;
171
172 // Register config routes
173 route_count = get_config_routes(&routes);
174 webserver_register_handler_array(s_server_handle, routes, route_count);
175
176 // Wildcard handler for embedded static files (must be last)
177 httpd_uri_t file_uri = {
178 .uri = "/*",
179 .method = HTTP_GET,
180 .handler = static_file_handler,
181 .user_ctx = NULL,
182 };
183 httpd_register_uri_handler(s_server_handle, &file_uri);
184
185 LOGI("Web server initialized successfully");
186 return s_server_handle;
187}

202esp_err_t webserver_register_handler(httpd_handle_t server,
203 const httpd_uri_t *uri_handler) {
204 if (server == NULL || uri_handler == NULL) {
205 return ESP_ERR_INVALID_ARG;
206 }
207
208 esp_err_t ret = httpd_register_uri_handler(server, uri_handler);
209 if (ret == ESP_OK) {
210 LOGI("Registered handler: %s [%d]", uri_handler->uri, uri_handler->method);
211 } else {
212 LOGE("Failed to register handler %s: %s", uri_handler->uri,
213 esp_err_to_name(ret));
214 }
215
216 return ret;
217}
