119static esp_err_t static_file_handler(httpd_req_t *req) {
120 const static_file_t *file = get_file_data(req->uri);
121
122 if (file == NULL) {
123 LOGW("File not found: %s", req->uri);
124 httpd_resp_send_404(req);
125 return ESP_OK;
126 }
127
128 httpd_resp_set_type(req, file->content_type);
129 if (file->content_encoding) {
130 httpd_resp_set_hdr(req, "Content-Encoding", file->content_encoding);
131 }
132
133 const size_t file_len = file->end - file->start;
134 httpd_resp_send(req, (const char *)file->start, file_len);
135
136 return ESP_OK;
137}

156httpd_handle_t webserver_start(const webserver_config_t *config) {
157 if (s_server_handle != NULL) {
158 LOGW("Web server already running");
159 return s_server_handle;
160 }
161
162 // Use provided config or defaults
163 uint16_t port = WEBSERVER_DEFAULT_PORT;
164 size_t max_handlers = WEBSERVER_DEFAULT_MAX_HANDLERS;
165 size_t stack_size = WEBSERVER_DEFAULT_STACK_SIZE;
166 UBaseType_t task_priority = WEBSERVER_DEFAULT_TASK_PRIORITY;
167
168 if (config) {
169 port = config->port;
170 max_handlers = config->max_uri_handlers;
171 stack_size = config->stack_size;
172 task_priority = config->task_priority;
173 }
174
175 // Create HTTP server configuration
176 httpd_config_t http_config = HTTPD_DEFAULT_CONFIG();
177 http_config.server_port = port;
178 http_config.max_uri_handlers = max_handlers;
179 http_config.stack_size = stack_size;
180 http_config.uri_match_fn = httpd_uri_match_wildcard;
181
182 // Start HTTP server
183 esp_err_t ret = httpd_start(&s_server_handle, &http_config);
184 if (ret != ESP_OK) {
185 LOGE("Failed to start HTTP server: %s", esp_err_to_name(ret));
186 s_server_handle = NULL;
187 return NULL;
188 }
189
190 LOGI("HTTP server started on port %d", port);
191
192 const httpd_uri_t *routes = NULL;
193 size_t route_count = 0;
194
195 // Register config routes
196 route_count = get_config_routes(&routes);
197 webserver_register_handler_array(s_server_handle, routes, route_count);
198
199 // Wildcard handler for static files from LittleFS (must be last)
200 httpd_uri_t file_uri = {
201 .uri = "/*",
202 .method = HTTP_GET,
203 .handler = static_file_handler,
204 .user_ctx = NULL,
205 };
206 httpd_register_uri_handler(s_server_handle, &file_uri);
207
208 // Create FreeRTOS task for the server
209 // This allows other tasks to continue running and makes the server
210 // async-ready
211 BaseType_t task_ret = xTaskCreate(webserver_task, "webserver", stack_size,
212 (void *)s_server_handle, task_priority,
213 &s_server_task_handle);
214
215 if (task_ret != pdPASS) {
216 LOGE("Failed to create web server task");
217 httpd_stop(s_server_handle);
218 s_server_handle = NULL;
219 return NULL;
220 }
221
222 LOGI("Web server initialized successfully");
223 return s_server_handle;
224}

243esp_err_t webserver_register_handler(httpd_handle_t server,
244 const httpd_uri_t *uri_handler) {
245 if (server == NULL || uri_handler == NULL) {
246 return ESP_ERR_INVALID_ARG;
247 }
248
249 esp_err_t ret = httpd_register_uri_handler(server, uri_handler);
250 if (ret == ESP_OK) {
251 LOGI("Registered handler: %s [%d]", uri_handler->uri, uri_handler->method);
252 } else {
253 LOGE("Failed to register handler %s: %s", uri_handler->uri,
254 esp_err_to_name(ret));
255 }
256
257 return ret;
258}
