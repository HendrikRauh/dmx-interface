17def _find_esp_port():
18 """Finds the first matching USB port based on known ESP VIDs/PIDs."""
19 # Known IDs for ESP boards and common USB-UART bridges
20 ESP_IDENTIFIERS = [
21 (0x303A, None), # Espressif (e.g., ESP32-S3/C3 native USB-JTAG-Serial)
22 (0x10C4, 0xEA60), # Silicon Labs CP210x (very common on ESP32)
23 (0x1A86, 0x7523), # WCH CH340 (common on budget ESP boards)
24 (0x0403, 0x6001), # FTDI FT232R
25 ]
26
27 ports = serial.tools.list_ports.comports()
28
29 for port in ports:
30 # Check if VID/PID match
31 for vid, pid in ESP_IDENTIFIERS:
32 if port.vid == vid and (pid is None or port.pid == pid):
33 return port.device
34
35 # Fallback to default if no matching device is found
36 return "/dev/ttyUSB0"

76def release(c):
77 """Build single binaries for release across all targets"""
78 version = os.environ.get("GITHUB_REF_NAME", "dev")
79
80 dist_dir = "dist"
81 if os.path.exists(dist_dir):
82 shutil.rmtree(dist_dir)
83 os.makedirs(dist_dir)
84
85 for board, defaults_file in TARGET_BOARDS.items():
86 print("\n=========================================")
87 if not os.path.exists(defaults_file):
88 print(f"⚠️ Warning: {defaults_file} not found. Skipping {board}...")
89 continue
90
91 print(f"🚀 Building Target: {board} ({defaults_file})")
92 print("=========================================")
93
94 c.run("idf.py fullclean", pty=True)
95 if os.path.exists("sdkconfig"):
96 os.remove("sdkconfig")
97
98 build_cmd = f"idf.py -D SDKCONFIG_DEFAULTS={defaults_file} -D CMAKE_BUILD_TYPE=Release build"
99 c.run(build_cmd, pty=True)
100
101 c.run("idf.py merge-bin", pty=True)
102
103 source_bin = "build/merged-binary.bin"
104 target_bin = os.path.join(dist_dir, f"ChaosDMX-{board}-{version}.bin")
105
106 if os.path.exists(source_bin):
107 shutil.copy(source_bin, target_bin)
108 print(f"➔ Asset successfully created: {target_bin}")
109 else:
110 print(f"❌ Error: {source_bin} was not created.")
111 raise Exit(code=1)
