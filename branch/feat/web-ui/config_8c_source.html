139esp_err_t config_init(void) {
140 if (s_is_initialized) {
141 LOGW("Component already initialized.");
142 return ESP_OK;
143 }
144
145 s_config_mutex = xSemaphoreCreateMutex();
146 if (s_config_mutex == NULL) {
147 LOGE("Failed to create configuration mutex.");
148 return ESP_ERR_NO_MEM;
149 }
150
151 nvs_handle_t handle;
152 esp_err_t err = nvs_open(NVS_NAMESPACE, NVS_READWRITE, &handle);
153 if (err != ESP_OK) {
154 if (err == ESP_ERR_NVS_NOT_FOUND) {
155 LOGW("NVS namespace missing. Staging factory defaults.");
156 load_factory_defaults();
157 } else {
158 LOGE("Failed to open NVS namespace '%s': %s", NVS_NAMESPACE,
159 esp_err_to_name(err));
160 vSemaphoreDelete(s_config_mutex);
161 s_config_mutex = NULL;
162 return err;
163 }
164 } else {
165 size_t size = sizeof(config_storage_t);
166 err = nvs_get_blob(handle, NVS_BLOB_KEY, &s_config, &size);
167
168 if (err == ESP_ERR_NVS_NOT_FOUND) {
169 LOGI("No configuration blob found. Initializing defaults.");
170 load_factory_defaults();
171 } else if (err != ESP_OK) {
172 LOGE("NVS read error: 0x%X. Falling back to defaults.", err);
173 load_factory_defaults();
174 } else if (size != sizeof(config_storage_t)) {
175 LOGW("Configuration size mismatch (%u vs %u). Resetting to defaults.",
176 (unsigned int)size, (unsigned int)sizeof(config_storage_t));
177 load_factory_defaults();
178 } else if (s_config.magic != APP_CONFIG_MAGIC) {
179 LOGW("Configuration magic mismatch (0x%08X). Data corrupted. Resetting.",
180 (unsigned int)s_config.magic);
181 load_factory_defaults();
182 } else if (s_config.version != APP_CONFIG_VERSION) {
183 LOGW("Configuration version mismatch (%u vs %u). Upgrading/Resetting.",
184 (unsigned int)s_config.version, (unsigned int)APP_CONFIG_VERSION);
185 load_factory_defaults();
186 } else {
187 LOGI("Configuration loaded successfully. Magic and Version match.");
188 s_is_dirty = false;
189 }
190 nvs_close(handle);
191 }
192
193 s_is_initialized = true;
194 return ESP_OK;
195}

207esp_err_t config_save(void) {
208 if (!s_is_initialized)
209 return ESP_ERR_NVS_NOT_INITIALIZED;
210
211 LOCK();
212 if (!s_is_dirty) {
213 UNLOCK();
214 LOGI("Flash save skipped: No changes detected.");
215 return ESP_OK;
216 }
217
218 nvs_handle_t handle;
219 esp_err_t err = nvs_open(NVS_NAMESPACE, NVS_READWRITE, &handle);
220 if (err != ESP_OK) {
221 UNLOCK();
222 LOGE("Failed to open NVS for saving: 0x%X", err);
223 return err;
224 }
225
226 err = nvs_set_blob(handle, NVS_BLOB_KEY, &s_config, sizeof(config_storage_t));
227 if (err == ESP_OK) {
228 err = nvs_commit(handle);
229 }
230
231 nvs_close(handle);
232
233 if (err == ESP_OK) {
234 s_is_dirty = false;
235 LOGI("Configuration successfully persisted to NVS.");
236 } else {
237 LOGE("Failed to write/commit config to NVS: 0x%X", err);
238 }
239
240 UNLOCK();
241 return err;
242}
