110static void network_event_handler(void *arg, esp_event_base_t event_base,
111 int32_t event_id, void *event_data) {
112 static uint8_t retry_count = 0;
113
114 if (event_base == WIFI_EVENT) {
115 switch (event_id) {
116 case WIFI_EVENT_AP_START:
117 esp_event_post(NETWORK_EVENT, NETWORK_EVENT_READY, NULL, 0, 0);
118 break;
119 case WIFI_EVENT_STA_START:
120 retry_count = 0;
121 esp_wifi_connect();
122 break;
123 case WIFI_EVENT_STA_DISCONNECTED: {
124 wifi_event_sta_disconnected_t *disconnected_event =
125 (wifi_event_sta_disconnected_t *)event_data;
126 LOGW("WiFi station disconnected. Reason code: %d",
127 disconnected_event->reason);
128
129 esp_event_post(NETWORK_EVENT, NETWORK_EVENT_DISCONNECTED, NULL, 0, 0);
130
131 if (reconnect_timer) {
132 // prevent multiple timers from running simultaneously
133 xTimerStop(reconnect_timer, 0);
134 }
135
136 switch (disconnected_event->reason) {
137 case WIFI_REASON_AUTH_FAIL:
138 case WIFI_REASON_NO_AP_FOUND_W_COMPATIBLE_SECURITY:
139 case WIFI_REASON_NO_AP_FOUND_IN_AUTHMODE_THRESHOLD:
140 case WIFI_REASON_802_1X_AUTH_FAILED:
141 case WIFI_REASON_PAIRWISE_CIPHER_INVALID:
142 case WIFI_REASON_GROUP_CIPHER_INVALID:
143 case WIFI_REASON_BAD_CIPHER_OR_AKM:
144 case WIFI_REASON_AKMP_INVALID:
145 case WIFI_REASON_CIPHER_SUITE_REJECTED:
146 case WIFI_REASON_UNSUPP_RSN_IE_VERSION:
147 case WIFI_REASON_INVALID_RSN_IE_CAP:
148 case WIFI_REASON_NOT_AUTHORIZED_THIS_LOCATION:
149 case WIFI_REASON_DISASSOC_SUPCHAN_BAD:
150 case WIFI_REASON_DISASSOC_PWRCAP_BAD:
151 LOGE("Critical configuration/security error. Reconnection halted.");
152 sta_reconnect = false;
153 esp_event_post(NETWORK_EVENT, NETWORK_EVENT_CONNECTION_FAILED, NULL, 0,
154 0);
155 break;
156 }
157
158 if (sta_reconnect) {
159 if (retry_count < MAX_RETRY_ATTEMPTS) {
160 retry_count++;
161
162 // exponential backoff delay: base * 2 ^(retry_count - 1), capped at
163 // RECONNECT_MAX_DELAY_MS
164 uint32_t delay_ms =
165 RECONNECT_BASE_DELAY_MS * (1 << (retry_count - 1));
166
167 if (delay_ms > RECONNECT_MAX_DELAY_MS) {
168 delay_ms = RECONNECT_MAX_DELAY_MS;
169 }
170
171 LOGI("Scheduling reconnect attempt %d/%d in %lu ms...", retry_count,
172 MAX_RETRY_ATTEMPTS, delay_ms);
173
174 if (reconnect_timer) {
175 // Also starts the timer if it is not already running
176 if (xTimerChangePeriod(reconnect_timer, pdMS_TO_TICKS(delay_ms),
177 0) != pdPASS) {
178 LOGE("Failed to change reconnect timer period");
179 }
180 }
181 } else {
182 LOGW("Max retry attempts reached. Giving up.");
183 sta_reconnect = false;
184 esp_event_post(NETWORK_EVENT, NETWORK_EVENT_CONNECTION_FAILED, NULL,
185 0, 0);
186 }
187 }
188 break;
189 }
190 case WIFI_EVENT_STA_STOP:
191 xTimerStop(reconnect_timer, 0);
192 esp_event_post(NETWORK_EVENT, NETWORK_EVENT_DISCONNECTED, NULL, 0, 0);
193 break;
194 case WIFI_EVENT_AP_STOP:
195 esp_event_post(NETWORK_EVENT, NETWORK_EVENT_DISCONNECTED, NULL, 0, 0);
196 break;
197 default:
198 break;
199 }
200 } else if (event_base == IP_EVENT) {
201 if (event_id == IP_EVENT_STA_GOT_IP) {
202 ip_event_got_ip_t *event = (ip_event_got_ip_t *)event_data;
203 LOGI("Got IP: " IPSTR, IP2STR(&event->ip_info.ip));
204 esp_event_post(NETWORK_EVENT, NETWORK_EVENT_READY, NULL, 0, 0);
205 retry_count = 0;
206 }
207 }
208}

225static esp_err_t validate_credentials(const char *ssid, const char *password) {
226 if (!ssid || strlen(ssid) == 0 || strlen(ssid) > 32) {
227 return ESP_ERR_INVALID_ARG;
228 }
229
230 const bool has_password = password && strlen(password) > 0;
231 if (has_password && (strlen(password) < 8 || strlen(password) > 63)) {
232 return ESP_ERR_INVALID_ARG;
233 }
234
235 return ESP_OK;
236}

251esp_err_t network_init() {
252 if (network_initialized) {
253 LOGW("Network is already initialized");
254 return ESP_ERR_INVALID_STATE;
255 }
256
257 if (network_mutex == NULL) {
258 network_mutex = xSemaphoreCreateMutex();
259 if (network_mutex == NULL) {
260 LOGE("Failed to create network mutex");
261 return ESP_ERR_NO_MEM;
262 }
263 }
264
265 if (reconnect_timer == NULL) {
266 reconnect_timer = xTimerCreate("wifi_reconnect_timer",
267 pdMS_TO_TICKS(RECONNECT_BASE_DELAY_MS),
268 pdFALSE, NULL, on_reconnect_timer);
269
270 if (reconnect_timer == NULL) {
271 LOGE("Failed to create reconnect timer");
272 return ESP_ERR_NO_MEM;
273 }
274 }
275
276 LOCK_NETWORK();
277
278 ESP_ERROR_CHECK(esp_netif_init());
279 esp_err_t err = esp_event_loop_create_default();
280 if (err != ESP_OK && err != ESP_ERR_INVALID_STATE) {
281 LOGE("Failed to create default event loop: %s", esp_err_to_name(err));
282 abort();
283 }
284
285 esp_netif_create_default_wifi_sta();
286 esp_netif_create_default_wifi_ap();
287
288 ESP_ERROR_CHECK(esp_event_handler_register(WIFI_EVENT, ESP_EVENT_ANY_ID,
289 network_event_handler, NULL));
290 ESP_ERROR_CHECK(esp_event_handler_register(IP_EVENT, ESP_EVENT_ANY_ID,
291 network_event_handler, NULL));
292
293
294 ESP_ERROR_CHECK(network_init_wifi());
295
296 network_initialized = true;
297 UNLOCK_NETWORK();
298 return ESP_OK;
299}

301esp_err_t network_start_ap(const char *ssid, const char *password) {
302 LOCK_NETWORK();
303
304 esp_err_t ret = ESP_OK;
305
306 if (!network_initialized) {
307 LOGW("Network is not initialized, cannot start WiFi AP");
308 ret = ESP_ERR_INVALID_STATE;
309 goto cleanup;
310 }
311
312 if (connection_started) {
313 LOGW("Network is already connected, cannot start WiFi AP");
314 ret = ESP_ERR_INVALID_STATE;
315 goto cleanup;
316 }
317
318 ESP_GOTO_ON_ERROR(validate_credentials(ssid, password), cleanup, LOG_TAG,
319 "Invalid WiFi credentials");
320
321 const bool has_password = password && strlen(password) > 0;
322 wifi_config_t wifi_config = {
323 .ap =
324 {
325 .channel = 0, // auto-select channel
326 .max_connection = 4,
327 .authmode = has_password ? WIFI_AUTH_WPA2_PSK : WIFI_AUTH_OPEN,
328 .pmf_cfg = {.required = false},
329 },
330 };
331
332 wifi_config.ap.ssid_len = strlen(ssid);
333 memcpy(wifi_config.ap.ssid, ssid, wifi_config.ap.ssid_len);
334 if (has_password) {
335 strlcpy((char *)wifi_config.ap.password, password,
336 sizeof(wifi_config.ap.password));
337 }
338
339 ESP_GOTO_ON_ERROR(esp_wifi_set_mode(WIFI_MODE_AP), cleanup, LOG_TAG,
340 "Failed to set WiFi mode to AP");
341 ESP_GOTO_ON_ERROR(esp_wifi_set_config(WIFI_IF_AP, &wifi_config), cleanup,
342 LOG_TAG, "Failed to set WiFi AP configuration");
343
344 ESP_GOTO_ON_ERROR(esp_wifi_start(), cleanup, LOG_TAG,
345 "Failed to start WiFi AP");
346 LOGI("WiFi AP started: SSID=%s", ssid);
347
348 connection_started = true;
349 connection_type = APP_CONFIG_CONN_WIFI_AP;
350
351cleanup:
352 UNLOCK_NETWORK();
353 return ret;
354}
