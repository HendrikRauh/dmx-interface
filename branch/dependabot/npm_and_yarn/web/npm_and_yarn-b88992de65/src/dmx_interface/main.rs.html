1//! DMX interface firmware for ESP32-S2 (Lolin S2 Mini).
2//!
3//! Current scope: status LED on GPIO7 (LEDC PWM, own embassy task), USB-CDC
4//! with esptool auto-reset, ring-buffer logging over the CDC port, and
5//! panic reports persisted across resets.
6
7#![no_std]
8#![no_main]
9#![deny(missing_docs)]
10#![allow(dead_code)]

36/// Clears the panic boot-loop counter once the firmware ran stably for 30 s.
37#[embassy_executor::task]
38async fn stable_marker() {
39    Timer::after(Duration::from_secs(30)).await;
40    panic_report::mark_stable();
41}
42
43/// Entry point.
44#[esp_rtos::main]
45async fn main(spawner: Spawner) -> ! {
46    let peripherals = esp_hal::init(esp_hal::Config::default().with_cpu_clock(CpuClock::max()));
47
48    // ── Heap allocator ─────────────────────────────────────────────────────
49    {
50        /// Heap size in bytes (32 KiB internal RAM — the panic-report replay
51        /// and NVS blob reads allocate, 8 KiB was too tight).
52        const HEAP_SIZE: usize = 32 * 1024;
53        /// Backing memory for the heap allocator.
54        static mut HEAP_MEM: [u8; HEAP_SIZE] = [0u8; HEAP_SIZE];
55        let heap_ptr = core::ptr::addr_of_mut!(HEAP_MEM).cast::<u8>();
56        // SAFETY: `heap_ptr` points to the whole `HEAP_MEM` array, which is
57        // valid for `'static`, exclusively owned by the allocator afterwards,
58        // handed over exactly once, and `size > 0`.
59        let region = unsafe {
60            HeapRegion::new(
61                heap_ptr,
62                HEAP_SIZE,
63                esp_alloc::MemoryCapability::Internal.into(),
64            )
65        };
66        // SAFETY: the heap region is registered exactly once at init time,
67        // before anything allocates.
68        unsafe {
69            esp_alloc::HEAP.add_region(region);
70        }
71    }
72
73    // ── Embassy scheduler ──────────────────────────────────────────────────
74    let timg0 = TimerGroup::new(peripherals.TIMG0);
75    let sw_int = SoftwareInterruptControl::new(peripherals.SW_INTERRUPT);
76    esp_rtos::start(timg0.timer0, sw_int.software_interrupt0);
77
78    // ── Logging + last-panic report ────────────────────────────────────────
79    // Must follow esp_rtos::start (uptime timestamps need the time driver)
80    // and precede the USB spawn, so a stored panic report is already in the
81    // ring when the first host connects.
82    logging::init();
83    storage::init(peripherals.FLASH);
84    // Load the persisted config and push it into every consumer (LED brightness).
85    storage::load().apply();
86    panic_report::report_last_panic();
87
88    // ── USB-CDC (esptool auto-reset + log drain) ───────────────────────────
89    spawner.spawn(
90        hardware::usb::task(peripherals.USB0, peripherals.GPIO20, peripherals.GPIO19)
91            .expect("USB task spawn failed"),
92    );
93    spawner.spawn(stable_marker().expect("stable marker spawn failed"));
94
95    // ── Status LED (LEDC PWM, own task) ─────────────────────────────────────
96    hardware::led::set(LedStatus::Startup);
97    spawner.spawn(
98        hardware::led::task(peripherals.LEDC, peripherals.GPIO7).expect("LED task spawn failed"),
99    );
100
101    log::info!("Init complete");
102
103    // TODO(phase 6): switch to Ok only once the network is up — the old
104    // firmware waited for NETWORK_READY + 2 s before LED_MODE_NORMAL.
105    // Until then, keep the Startup breathing visible for 3 s (three full
106    // breaths at the 1 s period) before settling into the steady Ok state.
107    Timer::after(Duration::from_secs(3)).await;
108    hardware::led::set(LedStatus::Ok);
109
110    // No main-loop duties yet — Phase 8 moves system work into tasks.
111    loop {
112        core::future::pending::<()>().await;
113    }
114}
