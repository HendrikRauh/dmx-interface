1//! DMX interface firmware for ESP32-S2 (Lolin S2 Mini).
2//!
3//! Current scope: LED effect on GPIO7 (LEDC PWM), USB-CDC with esptool
4//! auto-reset, and a diagnostic blink on the onboard LED (GPIO15).
5

44/// Entry point.
45#[esp_rtos::main]
46async fn main(spawner: Spawner) -> ! {
47    let peripherals = esp_hal::init(esp_hal::Config::default().with_cpu_clock(CpuClock::max()));
48
49    // ── Heap allocator ─────────────────────────────────────────────────────
50    {
51        /// Heap size in bytes (8 KiB internal RAM).
52        const HEAP_SIZE: usize = 8 * 1024;
53        /// Backing memory for the heap allocator.
54        static mut HEAP_MEM: [u8; HEAP_SIZE] = [0u8; HEAP_SIZE];
55        let heap_ptr = core::ptr::addr_of_mut!(HEAP_MEM).cast::<u8>();
56        // SAFETY: `heap_ptr` points to the whole `HEAP_MEM` array, which is
57        // valid for `'static`, exclusively owned by the allocator afterwards,
58        // handed over exactly once, and `size > 0`.
59        let region = unsafe {
60            HeapRegion::new(
61                heap_ptr,
62                HEAP_SIZE,
63                esp_alloc::MemoryCapability::Internal.into(),
64            )
65        };
66        // SAFETY: the heap region is registered exactly once at init time,
67        // before anything allocates.
68        unsafe {
69            esp_alloc::HEAP.add_region(region);
70        }
71    }
72
73    // ── Embassy scheduler ──────────────────────────────────────────────────
74    let timg0 = TimerGroup::new(peripherals.TIMG0);
75    let sw_int = SoftwareInterruptControl::new(peripherals.SW_INTERRUPT);
76    esp_rtos::start(timg0.timer0, sw_int.software_interrupt0);
77
78    // ── USB-CDC (esptool auto-reset) ───────────────────────────────────────
79    spawner.spawn(
80        hardware::usb::task(peripherals.USB0, peripherals.GPIO20, peripherals.GPIO19)
81            .expect("USB task spawn failed"),
82    );
83
84    // ── LEDC PWM (status LED) ──────────────────────────────────────────────
85    let mut ledc = Ledc::new(peripherals.LEDC);
86    ledc.set_global_slow_clock(LSGlobalClkSource::APBClk);
87
88    let mut timer0 = ledc.timer::<LowSpeed>(timer::Number::Timer0);
89    timer0
90        .configure(timer::config::Config {
91            duty: timer::config::Duty::Duty14Bit,
92            clock_source: LSClockSource::APBClk,
93            frequency: Rate::from_khz(1),
94        })
95        .expect("LEDC timer0 config failed");
96
97    let mut led = ledc.channel(channel::Number::Channel0, peripherals.GPIO7);
98    led.configure(channel::config::Config {
99        timer: &timer0,
100        duty_pct: 0,
101        drive_mode: esp_hal::gpio::DriveMode::PushPull,
102    })
103    .expect("LEDC channel0 config failed");
104
105    println!("Init complete — LED blink");
106
107    // Onboard LED (GPIO15): 1 Hz blink = main loop is alive (no serial out).
108    let mut diag = Output::new(peripherals.GPIO15, Level::Low, OutputConfig::default());
109    let mut diag_ticks = 0u32;
110
111    let effect = LedEffect::Blinking;
112    let start = Instant::now();
113
114    loop {
115        let elapsed = start.elapsed().as_millis();
116        hardware::led::apply(&led, &effect, 100, elapsed);
117        diag_ticks = diag_ticks.wrapping_add(1);
118        if diag_ticks.is_multiple_of(50) {
119            diag.toggle();
120        }
121        Timer::after(Duration::from_millis(10)).await;
122    }
123}
