1//! USB-CDC-ACM device with esptool auto-reset support.
2//!
3//! The Lolin S2 Mini has no USB-UART bridge, so esptool cannot toggle
4//! GPIO0/EN via physical DTR/RTS pins. Instead, the firmware exposes the
5//! native USB-OTG port as a CDC-ACM device and interprets the CDC
6//! control-line state the way the classic devkit reset sequence does:
7//!
8//! | DTR | RTS | Meaning            | Action                          |
9//! |-----|-----|--------------------|---------------------------------|
10//! | 1   | 0   | IO0 low            | set `FORCE_DOWNLOAD_BOOT`       |
11//! | 0   | 1   | EN low (reset)     | software reset                  |
12//! | 0   | 0   | idle               | clear `FORCE_DOWNLOAD_BOOT`     |
13//! | 1   | 1   | transient          | no-op                           |
14//!
15//! esptool's `--before usb-reset` walk is `(0,0) -> (1,0) -> (0,1) ->
16//! (0,0)`: the `(1,0)` state arms `FORCE_DOWNLOAD_BOOT`, then `(0,1)`
17//! resets into the ROM bootloader.
18//!
19//! The flag lives in the RTC domain and survives software resets. The ROM
20//! bootloader does **not** clear it on entry — after a flash the chip would
21//! stay stuck in the bootloader forever (until power-on). `inv flash`
22//! therefore clears the flag itself after flashing (esptool `write-reg
23//! 0x3F408128 0 0x1` through the stub) and then resets via the RTC watchdog,
24//! which boots the application.
25

42/// Espressif USB vendor ID (same as the ROM bootloader).
43const USB_VID: u16 = 0x303A;
44
45/// Espressif generic CDC-ACM product ID.
46const USB_PID: u16 = 0x3001;
47
48/// The S2 peripheral types are invariant in their lifetime, which pins the
49/// USB driver to `'static`; the embassy-usb buffers therefore must be
50/// `'static` too. They are initialized exactly once at task start.
51static EP_OUT_BUFFER: StaticCell<[u8; 1024]> = StaticCell::new();
52/// Bulk OUT endpoint buffer for the CDC data endpoint.
53static CONFIG_DESCRIPTOR: StaticCell<[u8; 256]> = StaticCell::new();
54/// Buffer for the generated USB configuration descriptor.
55static BOS_DESCRIPTOR: StaticCell<[u8; 256]> = StaticCell::new();
56/// Buffer for the generated USB BOS descriptor.
57static CONTROL_BUF: StaticCell<[u8; 64]> = StaticCell::new();
58/// Scratch buffer for USB control transfers.
59static CDC_STATE: StaticCell<State<'static>> = StaticCell::new();
60
61/// Runs the USB-CDC-ACM device and the esptool reset protocol.
62///
63/// The DP/DM pins are fixed by the S2 hardware (GPIO20/GPIO19).
64#[embassy_executor::task]
65pub async fn task(usb0: USB0<'static>, dp: GPIO20<'static>, dm: GPIO19<'static>) -> ! {
66    let usb = Usb::new(usb0, dp, dm);
67
68    let driver = Driver::new(usb, EP_OUT_BUFFER.init([0u8; 1024]), Config::default());
69
70    let mut config = embassy_usb::Config::new(USB_VID, USB_PID);
71    config.manufacturer = Some("dmx-interface");
72    config.product = Some("DMX-Interface USB-CDC");
73    config.device_class = 0xEF;
74    config.device_sub_class = 0x02;
75    config.device_protocol = 0x01;
76    config.composite_with_iads = true;
77
78    let mut builder = Builder::new(
79        driver,
80        config,
81        CONFIG_DESCRIPTOR.init([0u8; 256]),
82        BOS_DESCRIPTOR.init([0u8; 256]),
83        &mut [],
84        CONTROL_BUF.init([0u8; 64]),
85    );
86    let mut class = CdcAcmClass::new(&mut builder, CDC_STATE.init(State::new()), 64);
87    let mut device = builder.build();
88
89    println!("USB-CDC ready");
90    join(device.run(), control_monitor(&mut class)).await;
91    unreachable!()
92}

94/// Polls the DTR/RTS control lines and mirrors them onto the strapping
95/// semantics (see module docs).
96async fn control_monitor(class: &mut CdcAcmClass<'static, Driver<'static>>) -> ! {
97    let mut prev = (class.dtr(), class.rts());
98    loop {
99        Timer::after(Duration::from_millis(10)).await;
100        let now = (class.dtr(), class.rts());
101        if now == prev {
102            continue;
103        }
104        prev = now;
105        match now {
106            (false, false) => set_force_download(false),
107            (true, false) => set_force_download(true),
108            (false, true) => enter_bootloader(),
109            (true, true) => {}
110        }
111    }
112}

114/// Set or clear the ROM `FORCE_DOWNLOAD_BOOT` flag in `RTC_CNTL_OPTION1`.
115///
116/// The register lives in the RTC domain and survives a software reset, so
117/// the ROM bootloader sees it on the next boot.
118fn set_force_download(armed: bool) {
119    let rtc_cntl = unsafe { esp32s2::RTC_CNTL::steal() };
120    rtc_cntl
121        .options1()
122        .modify(|_, w| w.force_download_boot().bit(armed));
123}

125/// Resets into the ROM USB-CDC download mode (bootloader).
126///
127/// `(0,1)` is a plain reset: the boot target is decided by the
128/// `FORCE_DOWNLOAD_BOOT` flag, which the `(1,0)` state arms. esptool's
129/// `--before usb-reset` walk therefore lands in the bootloader. The flag is
130/// only cleared again by `inv flash` after flashing (esptool `write-reg`)
131/// — the ROM bootloader does not clear it itself.
132fn enter_bootloader() -> ! {
133    esp_hal::system::software_reset()
