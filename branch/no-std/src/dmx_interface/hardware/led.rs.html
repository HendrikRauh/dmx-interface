1//! Hardware abstraction for status LED effects.
2//!
3//! Provides PWM-based LED control via ESP-IDF LEDC peripheral,
4//! supporting various blinking and breathing patterns.
5
6use esp_hal::ledc::{
7    LowSpeed,
8    channel::{self, ChannelIFace},
9};
10
11/// Available LED effects for the status LED.
12///
13/// Each effect modulates the LED brightness over time using PWM duty cycle.
14pub enum LedEffect {
15    /// LED is off (0% duty cycle).
16    Off,
17    /// LED is on at a constant brightness.
18    Solid,
19    /// LED fades in and out with a triangular waveform (2s period).
20    Breathing,
21    /// LED toggles on/off every 500ms.
22    Blinking,
23    /// LED toggles on/off every 100ms.
24    FastBlink,
25}
26
27/// Apply the given LED effect to a LEDC channel.
28///
29/// # Arguments
30/// * `channel` - The LEDC PWM channel to control.
31/// * `effect` - The desired LED effect.
32/// * `brightness` - Maximum brightness level (0–255, scaled to 0–100% duty).
33/// * `elapsed_ms` - Milliseconds since the effect was started (for timing).
34pub fn apply(
35    channel: &mut channel::Channel<'_, LowSpeed>,
36    effect: &LedEffect,
37    brightness: u8,
38    elapsed_ms: u64,
39) {
40    match effect {
41        LedEffect::Off => {
42            channel.set_duty(0).ok();
43        }
44        LedEffect::Solid => {
45            channel.set_duty(brightness * 100 / 255).ok();
46        }
47        LedEffect::Breathing => {
48            let period = 2000u64;
49            let pos = elapsed_ms % period;
50            let t = if pos < period / 2 {
51                pos as f32 / (period as f32 / 2.0)
52            } else {
53                2.0 - pos as f32 / (period as f32 / 2.0)
54            };
55            let duty_pct = (t * brightness as f32 / 255.0 * 100.0) as u8;
56            channel.set_duty(duty_pct.min(100)).ok();
57        }
58        LedEffect::Blinking => {
59            let on = (elapsed_ms % 1000) < 500;
60            channel
61                .set_duty(if on { brightness * 100 / 255 } else { 0 })
62                .ok();
63        }
64        LedEffect::FastBlink => {
65            let on = (elapsed_ms % 200) < 100;
66            channel
67                .set_duty(if on { brightness * 100 / 255 } else { 0 })
68                .ok();
69        }
70    }
71}
