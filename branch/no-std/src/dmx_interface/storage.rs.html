1//! NVS persistent storage for device configuration.
2//!
3//! Wraps `esp-nvs` to provide config load/save/clear operations.
4//! Config is serialized as a binary blob via `postcard`.
5
6extern crate alloc;
7
8use alloc::vec::Vec;
9use esp_hal::peripherals::FLASH;
10use esp_nvs::{Key, Nvs};
11use esp_println::println;
12use esp_storage::FlashStorage;
13
14use crate::config::Config;
15
16/// NVS partition offset (default for ESP32-S2).
17const NVS_OFFSET: usize = 0x9000;
18
19/// NVS partition size (default for ESP32-S2: 24 KiB).
20const NVS_SIZE: usize = 0x6000;
21
22/// NVS namespace for config storage.
23const NS_CONFIG: Key = Key::from_str("config");
24
25/// NVS key for the config blob.
26const KEY_CFG: Key = Key::from_str("cfg");
27
28/// Persistent storage handle wrapping esp-nvs.
29pub struct ConfigStorage<'a> {
30    nvs: Nvs<FlashStorage<'a>>,
31}
32
33impl ConfigStorage<'_> {
34    /// Initialize NVS storage.
35    ///
36    /// Must be called once at startup before any load/save operations.
37    /// Takes ownership of the FLASH peripheral (not used elsewhere).
38    pub fn new(flash: FLASH<'static>) -> ConfigStorage<'static> {
39        let flash = FlashStorage::new(flash);
40        let nvs = Nvs::new(NVS_OFFSET, NVS_SIZE, flash).expect("NVS init failed");
41        ConfigStorage { nvs }
42    }
43
44    /// Load config from NVS.
45    ///
46    /// Returns `Config::default()` if no config is stored or if deserialization fails.
47    pub fn load(&mut self) -> Config {
48        match self.nvs.get::<Vec<u8>>(&NS_CONFIG, &KEY_CFG) {
49            Ok(bytes) => match Config::from_bytes(&bytes) {
50                Ok(config) => {
51                    println!("Config loaded from NVS ({} bytes)", bytes.len());
52                    config
53                }
54                Err(_e) => {
55                    println!("Config deserialization failed: {_e:?}, using defaults");
56                    Config::default()
57                }
58            },
59            Err(_e) => {
60                println!("NVS read failed: {_e:?}, using defaults");
61                Config::default()
62            }
63        }
64    }
65
66    /// Save config to NVS.
67    ///
68    /// Serializes the config to bytes and writes it as a blob.
69    pub fn save(&mut self, config: &Config) {
70        match config.to_bytes() {
71            Ok(bytes) => match self.nvs.set(&NS_CONFIG, &KEY_CFG, bytes.as_slice()) {
72                Ok(()) => println!("Config saved to NVS ({} bytes)", bytes.len()),
73                Err(_e) => println!("NVS write failed: {_e:?}"),
74            },
75            Err(_e) => println!("Config serialization failed: {_e:?}"),
76        }
77    }
78
79    /// Clear all config data from NVS (reverts to defaults on next load).
80    pub fn clear(&mut self) {
81        match self.nvs.delete(&NS_CONFIG, &KEY_CFG) {
82            Ok(()) => println!("Config cleared from NVS"),
83            Err(_e) => println!("NVS clear failed: {_e:?}"),
84        }
85    }
86}
